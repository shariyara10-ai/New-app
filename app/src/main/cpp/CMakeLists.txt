cmake_minimum_required(VERSION 3.22)
project(CineZehnWhisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/whisper/CMakeLists.txt")
  message(FATAL_ERROR "whisper.cpp sources missing; use GitHub Actions clone step")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/whisper" "${CMAKE_CURRENT_BINARY_DIR}/whisper" EXCLUDE_FROM_ALL)
add_library(cinezehn_whisper SHARED "whisper_bridge.cpp")
target_include_directories(cinezehn_whisper PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/whisper/include")
find_library(log-lib log)
target_link_libraries(cinezehn_whisper PRIVATE whisper ${log-lib})
